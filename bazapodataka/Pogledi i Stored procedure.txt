RESENJE - POGLEDI
------------------------------------------
CREATE VIEW `Fakultet_VP_2025`.`SviPodaciOKnjigama` AS select `knjiga`.`ISBN`, `knjiga`.`Naziv`, `knjiga`.`Autor`, `zanr`.`Naziv` as `NazivZanra` from `Fakultet_VP_2025`.`knjiga` INNER JOIN `Fakultet_VP_2025`.`zanr` ON `Fakultet_VP_2025`.`knjiga`.`OznakaZanra`=`Fakultet_VP_2025`.`zanr`.`Oznaka`;


CREATE VIEW `Fakultet_VP_2025`.`SviPodaciOKnjigamaSaSlikom` AS select `knjiga`.`ISBN`, `knjiga`.`Naziv`, `knjiga`.`Autor`, `zanr`.`Naziv` as `NazivZanra`, `knjiga`.`NazivFajlaSlike`   from `Fakultet_VP_2025`.`knjiga` INNER JOIN `Fakultet_VP_2025`.`zanr` ON `Fakultet_VP_2025`.`knjiga`.`OznakaZanra`=`Fakultet_VP_2025`.`zanr`.`Oznaka`;


RESENJE ZA STORED PROCEDURU- insert into
-----------------------
USE `Fakultet_VP_2025`;
DROP procedure IF EXISTS `DodajKnjigu`;
DELIMITER $$
USE `Fakultet_VP_2025`$$
CREATE PROCEDURE `DodajKnjigu` (IN ISBNParametar varchar(13), IN NazivParametar varchar(100), IN AutorParametar varchar(100), IN OznakaZanraParametar char(2), IN NazivFajlaSlikeParametar varchar(100))
BEGIN
INSERT INTO `knjiga` (`ISBN`, `Naziv`, `Autor`, `OznakaZanra`, `NazivFajlaSlike`) VALUES (ISBNParametar, NazivParametar, AutorParametar, OznakaZanraParametar, NazivFajlaSlikeParametar);
END
$$
DELIMITER ;
